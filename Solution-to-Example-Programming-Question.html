num = int(input("Enter how many Fibonacci numbers you want: "))
print("  Sequence: ", end="")

first = 0
second = 1
for i in range(num):
  if i <= 1:
    next = i
  else:
    next = first + second
    first = second
    second = next
  print(next, end=" ")

print("...")
